# Only one product executable. CSV/TXT/value and GUI/TUI are runtime modes.
add_executable(quanta main.c)
quanta_target(quanta)
set_target_properties(quanta PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
target_include_directories(quanta PRIVATE ${PROJECT_SOURCE_DIR}/frontends/private)
target_link_libraries(quanta PRIVATE quanta_frontends quanta_sensors)
if(WIN32)
    target_sources(quanta PRIVATE windows_entry.c ${PROJECT_SOURCE_DIR}/resources/quanta.rc)
    target_include_directories(quanta PRIVATE ${PROJECT_SOURCE_DIR}/resources ${PROJECT_SOURCE_DIR}/osal/private)
    if(MINGW)
        target_link_options(quanta PRIVATE -municode -Wl,--nxcompat -Wl,--dynamicbase)
    else()
        target_link_options(quanta PRIVATE /DYNAMICBASE /NXCOMPAT /MANIFEST:NO)
    endif()
endif()
