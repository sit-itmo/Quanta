# sensors/CMakeLists.txt
# Source list and settings for this module/platform; no runtime logic.
# Common algorithms/HAL metrics are separate from OS snapshots and SDK code.
add_library(quanta_sensors STATIC registry.c metric_docs.c common/runtime.c)
quanta_target(quanta_sensors)
target_include_directories(quanta_sensors PRIVATE private ${PROJECT_SOURCE_DIR}/metrics)
target_link_libraries(quanta_sensors PUBLIC quanta_core)
if(QUANTA_PORTABLE)
    return()
endif()
target_sources(quanta_sensors PRIVATE common/normalize.c os/os_metrics.c os/${QUANTA_PLATFORM}/os.c)
foreach(feature MEMORY CPU DMI NETWORK HWMON CPUID FILESYSTEM PROCESS)
    if(NOT QUANTA_WITH_${feature})
        continue()
    endif()
    if(feature STREQUAL "CPUID")
        if(QUANTA_ARCH_X86)
            target_sources(quanta_sensors PRIVATE arch/x86/metrics.c
                arch/x86/${QUANTA_PLATFORM}.c arch/x86/affinity_${QUANTA_PLATFORM}.c)
        endif()
        continue()
    endif()
    string(TOLOWER ${feature} group)
    if(feature STREQUAL "DMI")
        set(group board)
    elseif(feature STREQUAL "HWMON")
        set(group temperature)
    endif()
    target_sources(quanta_sensors PRIVATE os/${group}_metrics.c os/${QUANTA_PLATFORM}/${group}.c)
endforeach()
if(QUANTA_WITH_DMI)
    target_sources(quanta_sensors PRIVATE common/smbios.c)
    if(QUANTA_ARCH_X86)
        target_sources(quanta_sensors PRIVATE common/smbios_memory.c)
    endif()
endif()
if(WIN32)
    target_include_directories(quanta_sensors PRIVATE ${PROJECT_SOURCE_DIR}/osal/private)
    target_sources(quanta_sensors PRIVATE os/windows/cpu_instance.c os/windows/system_drive.c)
    target_link_libraries(quanta_sensors PRIVATE advapi32)
    if(QUANTA_WITH_MEMORY OR QUANTA_WITH_PROCESS)
        target_link_libraries(quanta_sensors PRIVATE psapi)
    endif()
    if(QUANTA_WITH_NETWORK)
        target_link_libraries(quanta_sensors PRIVATE iphlpapi)
    endif()
    if(QUANTA_WITH_CPU OR QUANTA_WITH_HWMON)
        target_sources(quanta_sensors PRIVATE os/windows/pdh_set.c)
        target_link_libraries(quanta_sensors PRIVATE pdh)
    endif()
endif()
