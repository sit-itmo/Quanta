# Registry/provider adapters and only the selected native HAL implementations.
add_library(quanta_sensors STATIC registry.c metric_docs.c
    ${PROJECT_SOURCE_DIR}/hal/common/windows_math.c
    os/provider.c ${PROJECT_SOURCE_DIR}/hal/${QUANTA_PLATFORM}/os.c)
quanta_target(quanta_sensors)
target_include_directories(quanta_sensors PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(quanta_sensors PRIVATE private
    ${PROJECT_SOURCE_DIR}/hal/include ${PROJECT_SOURCE_DIR}/hal/common
    ${PROJECT_SOURCE_DIR}/metrics)
target_link_libraries(quanta_sensors PUBLIC quanta_core)

foreach(feature MEMORY CPU DMI NETWORK HWMON CPUID FILESYSTEM PROCESS)
    if(NOT QUANTA_WITH_${feature})
        continue()
    endif()
    if(feature STREQUAL "CPUID" AND NOT QUANTA_ARCH_X86)
        continue()
    endif()
    string(TOLOWER ${feature} native)
    set(sensor ${native})
    if(feature STREQUAL "DMI")
        set(sensor board)
    elseif(feature STREQUAL "HWMON")
        set(sensor temperature)
    endif()
    target_sources(quanta_sensors PRIVATE ${sensor}/provider.c
        ${PROJECT_SOURCE_DIR}/hal/${QUANTA_PLATFORM}/${native}.c)
endforeach()
if(QUANTA_WITH_DMI)
    target_sources(quanta_sensors PRIVATE ${PROJECT_SOURCE_DIR}/hal/common/smbios.c)
endif()
if(WIN32)
    target_link_libraries(quanta_sensors PRIVATE advapi32)
    if(QUANTA_WITH_MEMORY OR QUANTA_WITH_PROCESS)
        target_link_libraries(quanta_sensors PRIVATE psapi)
    endif()
    if(QUANTA_WITH_NETWORK)
        target_link_libraries(quanta_sensors PRIVATE iphlpapi)
    endif()
    if(QUANTA_WITH_CPU OR QUANTA_WITH_HWMON)
        target_sources(quanta_sensors PRIVATE ${PROJECT_SOURCE_DIR}/hal/windows/pdh_set.c)
        target_link_libraries(quanta_sensors PRIVATE pdh)
    endif()
endif()
