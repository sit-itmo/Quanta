# Each metric implementation owns its native acquisition code. HAL is optional.
add_library(quanta_sensors STATIC registry.c metric_docs.c windows_math.c
    os/metrics.c os/${QUANTA_PLATFORM}.c)
quanta_target(quanta_sensors)
target_include_directories(quanta_sensors PUBLIC include)
target_include_directories(quanta_sensors PRIVATE private ${PROJECT_SOURCE_DIR}/metrics)
target_link_libraries(quanta_sensors PUBLIC quanta_core)

foreach(feature MEMORY CPU DMI NETWORK HWMON CPUID FILESYSTEM PROCESS)
    if(NOT QUANTA_WITH_${feature})
        continue()
    endif()
    if(feature STREQUAL "CPUID" AND NOT QUANTA_ARCH_X86)
        continue()
    endif()
    string(TOLOWER ${feature} sensor)
    if(feature STREQUAL "DMI")
        set(sensor board)
    elseif(feature STREQUAL "HWMON")
        set(sensor temperature)
    endif()
    target_sources(quanta_sensors PRIVATE ${sensor}/metrics.c ${sensor}/${QUANTA_PLATFORM}.c)
endforeach()
if(QUANTA_WITH_DMI)
    target_sources(quanta_sensors PRIVATE board/smbios.c)
    if(QUANTA_ARCH_X86)
        target_sources(quanta_sensors PRIVATE board/physical.c)
    endif()
endif()
if(QUANTA_WITH_CPUID AND QUANTA_ARCH_X86)
    target_sources(quanta_sensors PRIVATE cpuid/affinity_${QUANTA_PLATFORM}.c)
endif()
if(WIN32)
    target_link_libraries(quanta_sensors PRIVATE advapi32)
    if(QUANTA_WITH_MEMORY OR QUANTA_WITH_PROCESS)
        target_link_libraries(quanta_sensors PRIVATE psapi)
    endif()
    if(QUANTA_WITH_NETWORK)
        target_link_libraries(quanta_sensors PRIVATE iphlpapi)
    endif()
    if(QUANTA_WITH_CPU OR QUANTA_WITH_HWMON)
        target_sources(quanta_sensors PRIVATE pdh_set.c)
        target_link_libraries(quanta_sensors PRIVATE pdh)
    endif()
endif()
