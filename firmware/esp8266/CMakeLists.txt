cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Set IDF_PATH to ESP8266_RTOS_SDK (not ESP32 ESP-IDF). See docs/ESP8266.md.")
endif()
if(NOT EXISTS "$ENV{IDF_PATH}/components/esp8266")
    message(FATAL_ERROR "This firmware requires ESP8266_RTOS_SDK, not the ESP32 SDK.")
endif()
include("$ENV{IDF_PATH}/tools/cmake/project.cmake")
project(quanta_esp8266)
