# CMakeLists.txt
# Root configuration: one executable, platform features, frontends and native tests.
cmake_minimum_required(VERSION 3.21)
project(Quanta VERSION 0.8.0 LANGUAGES C)

# A plain single-configuration build is usable without selecting a preset.
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(QUANTA_PORTABLE "Build only the portable contract and common metrics (POSIX host)" OFF)
if(QUANTA_PORTABLE)
    if(WIN32)
        message(FATAL_ERROR "The supplied portable OSAL uses POSIX clocks/pthreads; use windows-native on Windows.")
    endif()
    set(QUANTA_PLATFORM portable)
elseif(WIN32)
    set(QUANTA_PLATFORM windows)
    set(QUANTA_OS_WINDOWS ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(QUANTA_PLATFORM linux)
    set(QUANTA_OS_LINUX ON)
else()
    message(FATAL_ERROR "Select QUANTA_PORTABLE=ON or add a native platform backend.")
endif()
if(MSVC)
    # No vcpkg packages or app-local copying are used by this self-contained tree.
    list(APPEND CMAKE_VS_GLOBALS "VcpkgEnabled=false" "VcpkgApplocalDeps=false")
endif()
if(MSVC AND MSVC_VERSION LESS 1928)
    message(FATAL_ERROR "A C11 compiler is required; use Visual Studio 2022.")
endif()

foreach(feature MEMORY CPU DMI NETWORK HWMON CPUID FILESYSTEM PROCESS)
    option(QUANTA_WITH_${feature} "Compile ${feature} metrics" ON)
    if(QUANTA_PORTABLE)
        set(QUANTA_WITH_${feature} OFF)
    endif()
endforeach()
option(QUANTA_EMBED_DOCS "Embed metric Markdown documentation" ON)
option(QUANTA_STRICT_STATIC "Static executable/CRT (Windows system DLLs remain)" OFF)
option(QUANTA_SANITIZE "AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(QUANTA_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(QUANTA_TESTS_ON_BUILD "Run CTest on every ordinary native build" ON)
option(QUANTA_PYTHON_TESTS "Enable additional Python checks when an interpreter is available" ON)
include(CTest)
if(QUANTA_SANITIZE AND (WIN32 OR QUANTA_STRICT_STATIC))
    message(FATAL_ERROR "Sanitizers require a non-static Linux/POSIX build.")
endif()
if(MSVC AND QUANTA_STRICT_STATIC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|x86|i[3-6]86)$" OR
   CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|Win32)$")
    set(QUANTA_ARCH_X86 ON)
endif()
if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM")
    set(QUANTA_ARCH_X86 OFF)
endif()

include(cmake/Frontends.cmake)
# All indirect C++ consumers (app and tests) must share this language scope.
# Enabling CXX only inside frontends/ loses its runtime on sibling C targets.
if(QUANTA_WITH_GUI)
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 11)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
endif()
if(WIN32)
    enable_language(RC)
endif()
configure_file(cmake/quanta_config.h.in generated/quanta_config.h)
include(cmake/CompilerOptions.cmake)
add_subdirectory(core)
add_subdirectory(sensors)
add_subdirectory(frontends)
add_subdirectory(app)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

include(GNUInstallDirs)
install(TARGETS quanta RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS quanta_core quanta_sensors quanta_frontends ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(TARGET imgui_vendor)
    install(TARGETS imgui_vendor ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
foreach(extra quanta_framebuffer quanta_imgui_opengl glfw)
    if(TARGET ${extra})
        get_target_property(imported ${extra} IMPORTED)
        if(NOT imported)
            install(TARGETS ${extra} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
        endif()
    endif()
endforeach()
install(DIRECTORY include/quanta DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/quanta/examples)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR})
message(STATUS "Quanta ${PROJECT_VERSION}: ${QUANTA_PLATFORM}; one executable: quanta")
message(STATUS "Frontends: file=ON, TUI=${QUANTA_WITH_TUI}, GUI=${QUANTA_WITH_GUI}")

if(QUANTA_WITH_GUI)
    message(STATUS "GUI backend: ${QUANTA_GUI_BACKEND}; no network downloads")
endif()
