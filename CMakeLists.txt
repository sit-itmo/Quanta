cmake_minimum_required(VERSION 3.21)
project(Quanta VERSION 0.4.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(WIN32)
    set(QUANTA_PLATFORM windows)
    set(QUANTA_OS_WINDOWS ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(QUANTA_PLATFORM linux)
    set(QUANTA_OS_LINUX ON)
else()
    message(FATAL_ERROR "This desktop baseline supports Linux and Windows only.")
endif()
if(MSVC AND MSVC_VERSION LESS 1928)
    message(FATAL_ERROR "Use Visual Studio 2019 16.8 or newer; VS 2022 is recommended.")
endif()

option(QUANTA_WITH_GUI "Build the optional Dear ImGui frontend" OFF)
option(QUANTA_WITH_TUI "Build the terminal frontend" ON)
foreach(feature MEMORY CPU DMI NETWORK HWMON CPUID FILESYSTEM PROCESS)
    option(QUANTA_WITH_${feature} "Compile the ${feature} metric implementation" ON)
endforeach()
option(QUANTA_EMBED_DOCS "Embed metric Markdown documentation" ON)
option(QUANTA_STRICT_STATIC "Static executable/CRT (Windows system DLLs remain)" OFF)
option(QUANTA_SANITIZE "Linux AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(QUANTA_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(QUANTA_TESTS_ON_BUILD "Run CTest on every ordinary native build" ON)
include(CTest)

if(QUANTA_SANITIZE AND (WIN32 OR QUANTA_STRICT_STATIC))
    message(FATAL_ERROR "Sanitizers require Linux and QUANTA_STRICT_STATIC=OFF.")
endif()
if(MSVC AND QUANTA_STRICT_STATIC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|x86|i[3-6]86)$" OR
   CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|Win32)$")
    set(QUANTA_ARCH_X86 ON)
endif()
if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM")
    set(QUANTA_ARCH_X86 OFF)
endif()
configure_file(cmake/quanta_config.h.in generated/quanta_config.h)
include(cmake/CompilerOptions.cmake)

add_subdirectory(hal)
add_subdirectory(core)
add_subdirectory(sensors)
add_subdirectory(frontends)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

include(GNUInstallDirs)
install(TARGETS quanta quanta_cli quanta_csv quanta_txt RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS quanta_core quanta_sensors quanta_hal ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY core/include/ osal/include/ sensors/include/ hal/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/quanta/examples)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR} PATTERN baseline EXCLUDE)
message(STATUS "Quanta ${PROJECT_VERSION}: ${QUANTA_PLATFORM}, GUI=${QUANTA_WITH_GUI}, TUI=${QUANTA_WITH_TUI}")
