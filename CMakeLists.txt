cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(inv VERSION 0.2.2 LANGUAGES C)
if(WIN32)
  set(INV_OS_WINDOWS ON)
  set(INV_OS_LINUX OFF)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(INV_OS_WINDOWS OFF)
  set(INV_OS_LINUX ON)
else()
  message(FATAL_ERROR "INV 0.2 implements Windows and Linux. Add a platform port for other systems.")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  if(MSVC_VERSION LESS 1928)
    message(FATAL_ERROR "Use Visual Studio 2019 16.8+ (recommended: VS2022 v143). Old v141/v141_xp lacks the required C frontend.")
  endif()
  set(CMAKE_C_STANDARD 11) # MSVC's conforming C frontend also implements our C99 subset.
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "INV libraries are always static" FORCE)
option(INV_WITH_GUI "Dear ImGui, software rasterizer, native window backend" OFF)
option(INV_WITH_TUI "Native console TUI; ANSI/termios on Linux" ON)
foreach(feature MEMORY CPU DMI NETWORK HWMON CPUID FILESYSTEM PROCESS)
  option(INV_WITH_${feature} "Enable ${feature} provider" ON)
endforeach()
option(INV_EMBED_DOCS "Embed metric documentation" ON)
option(INV_STRICT_STATIC "Linux: static ELF. Windows: static application/CRT; system DLLs remain." ON)
option(INV_SANITIZE "Linux AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
option(INV_BUILD_TESTS "Build tests" ON)
option(INV_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
if(INV_SANITIZE AND (INV_STRICT_STATIC OR WIN32))
  message(FATAL_ERROR "INV_SANITIZE requires Linux and INV_STRICT_STATIC=OFF.")
endif()
if(MSVC AND INV_STRICT_STATIC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|x86|i[3-6]86)$" OR
   CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|Win32)$")
  set(INV_ARCH_X86 ON)
else()
  set(INV_ARCH_X86 OFF)
endif()
# Generator architecture overrides a host architecture (e.g. an ARM64 target).
if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM")
  set(INV_ARCH_X86 OFF)
endif()
configure_file(cmake/inv_config.h.in generated/inv_config.h)
add_library(inv_build_flags INTERFACE)
target_include_directories(inv_build_flags INTERFACE
  ${PROJECT_SOURCE_DIR}/include ${PROJECT_SOURCE_DIR}/src/core
  ${PROJECT_SOURCE_DIR}/src/platform ${PROJECT_SOURCE_DIR}/metrics
  ${PROJECT_BINARY_DIR}/generated)
if(WIN32)
  target_compile_definitions(inv_build_flags INTERFACE
    WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
    _WIN32_WINNT=0x0601 WINVER=0x0601 NTDDI_VERSION=0x06010000
    PSAPI_VERSION=1 _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_definitions(inv_build_flags INTERFACE _POSIX_C_SOURCE=200809L _GNU_SOURCE)
  find_package(Threads REQUIRED)
endif()
if(MSVC)
  target_compile_options(inv_build_flags INTERFACE /W4 /utf-8 /Zc:preprocessor)
  if(INV_WARNINGS_AS_ERRORS)
    target_compile_options(inv_build_flags INTERFACE /WX)
  endif()
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(inv_build_flags INTERFACE -Wall -Wextra -Wpedantic -Wformat=2 -Wformat-security)
  if(INV_WARNINGS_AS_ERRORS)
    target_compile_options(inv_build_flags INTERFACE -Werror)
  endif()
  if(INV_SANITIZE)
    target_compile_options(inv_build_flags INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(inv_build_flags INTERFACE -fsanitize=address,undefined)
  endif()
  if(INV_STRICT_STATIC)
    target_link_options(inv_build_flags INTERFACE -static)
  endif()
endif()
if(WIN32)
  set(provider_dir src/providers/windows)
  set(platform_sources src/platform/windows.c src/platform/thread_windows.c)
else()
  set(provider_dir src/providers)
  set(platform_sources src/platform/linux.c src/platform/thread_posix.c)
endif()
set(providers ${provider_dir}/os.c)
foreach(feature MEMORY CPU DMI NETWORK HWMON FILESYSTEM PROCESS)
  if(INV_WITH_${feature})
    string(TOLOWER "${feature}" lower)
    list(APPEND providers ${provider_dir}/${lower}.c)
  endif()
endforeach()
if(INV_WITH_CPUID AND INV_ARCH_X86)
  list(APPEND providers ${provider_dir}/cpuid.c)
endif()
if(WIN32 AND (INV_WITH_CPU OR INV_WITH_HWMON))
  list(APPEND providers src/providers/windows/pdh_set.c)
endif()
add_library(inv_core STATIC src/core/common.c src/core/registry.c src/core/engine.c
  src/core/docs.c src/query/query.c ${platform_sources} ${providers}
  src/providers/common/smbios.c src/providers/common/windows_math.c)
target_include_directories(inv_core PUBLIC src/providers/common)
target_link_libraries(inv_core PUBLIC inv_build_flags)
if(WIN32)
  target_link_libraries(inv_core PUBLIC advapi32)
  if(INV_WITH_MEMORY OR INV_WITH_PROCESS)
    target_link_libraries(inv_core PUBLIC psapi)
  endif()
  if(INV_WITH_NETWORK)
    target_link_libraries(inv_core PUBLIC iphlpapi)
  endif()
  if(INV_WITH_CPU OR INV_WITH_HWMON)
    target_link_libraries(inv_core PUBLIC pdh)
  endif()
else()
  target_link_libraries(inv_core PUBLIC Threads::Threads m)
endif()
add_library(inv_view STATIC src/view/model.c src/view/live.c)
target_link_libraries(inv_view PUBLIC inv_core)
add_library(inv_file STATIC src/frontends/file/file.c)
target_link_libraries(inv_file PUBLIC inv_view)
add_executable(inv src/frontends/cli/main.c)
target_link_libraries(inv PRIVATE inv_file)
if(WIN32)
  enable_language(RC)
  target_sources(inv PRIVATE src/frontends/cli/windows_entry.c resources/inv.rc)
  target_include_directories(inv PRIVATE resources)
  if(MINGW)
    target_compile_definitions(inv_build_flags INTERFACE __USE_MINGW_ANSI_STDIO=1)
    target_link_options(inv PRIVATE -municode -Wl,--nxcompat -Wl,--dynamicbase)
  else()
    target_link_options(inv PRIVATE /DYNAMICBASE /NXCOMPAT /MANIFEST:NO)
  endif()
endif()
if(INV_WITH_TUI)
  if(WIN32)
    set(terminal_src src/frontends/tui/terminal_windows.c)
  else()
    set(terminal_src src/frontends/tui/terminal_posix.c)
  endif()
  add_library(inv_tui STATIC src/frontends/tui/tui.c ${terminal_src})
  target_link_libraries(inv_tui PUBLIC inv_view)
  target_link_libraries(inv PRIVATE inv_tui)
endif()
if(INV_WITH_GUI)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 11)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(cmake/ImGui.cmake)
  if(WIN32)
    set(window_src src/frontends/imgui/win32_platform.c)
  else()
    set(window_src src/frontends/imgui/xcb_platform.c)
    find_path(INV_XCB_INCLUDE_DIR xcb/xcb.h REQUIRED)
    set(saved_suffixes ${CMAKE_FIND_LIBRARY_SUFFIXES})
    set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
    find_library(INV_XCB_LIBRARY NAMES xcb REQUIRED)
    set(CMAKE_FIND_LIBRARY_SUFFIXES ${saved_suffixes})
  endif()
  add_library(inv_gui STATIC src/frontends/imgui/gui.cpp ${window_src} src/frontends/imgui/raster.c)
  target_include_directories(inv_gui PRIVATE src/frontends/imgui)
  target_link_libraries(inv_gui PUBLIC inv_view imgui_vendor)
  if(WIN32)
    target_link_libraries(inv_gui PUBLIC user32 gdi32 imm32)
  else()
    target_include_directories(inv_gui PRIVATE ${INV_XCB_INCLUDE_DIR})
    target_link_libraries(inv_gui PUBLIC ${INV_XCB_LIBRARY})
  endif()
  target_link_libraries(inv PRIVATE inv_gui)
endif()
if(INV_BUILD_TESTS)
  enable_testing()
  add_executable(inv_tests tests/test_core.c src/frontends/imgui/raster.c)
  target_include_directories(inv_tests PRIVATE src/frontends/imgui)
  target_link_libraries(inv_tests PRIVATE inv_view)
  add_test(NAME unit COMMAND inv_tests)
  add_executable(inv_windows_data_tests tests/test_windows_data.c)
  target_link_libraries(inv_windows_data_tests PRIVATE inv_core)
  add_test(NAME windows_data COMMAND inv_windows_data_tests)
  if(NOT WIN32)
    add_executable(inv_device_tests tests/test_devices.c)
    target_link_libraries(inv_device_tests PRIVATE inv_core)
    add_test(NAME devices COMMAND inv_device_tests)
  else()
    add_executable(inv_windows_tests tests/test_windows.c)
    target_link_libraries(inv_windows_tests PRIVATE inv_view)
    add_test(NAME windows_platform COMMAND inv_windows_tests)
  endif()
  if(NOT WIN32)
    # Compile the Windows table and query validator on the host. No fake WinAPI:
    # providers are address-only stubs in a separate, non-installed test target.
    set(INV_OS_WINDOWS ON)
    set(INV_OS_LINUX OFF)
    set(saved_gui ${INV_WITH_GUI})
    set(saved_tui ${INV_WITH_TUI})
    set(INV_WITH_GUI ON)
    set(INV_WITH_TUI ON)
    configure_file(cmake/inv_config.h.in schema-config/inv_config.h)
    set(INV_WITH_GUI ${saved_gui})
    set(INV_WITH_TUI ${saved_tui})
    set(INV_OS_WINDOWS OFF)
    set(INV_OS_LINUX ON)
    add_executable(inv_windows_schema_tests tests/test_windows_schema.c
      tests/windows_schema_stubs.c src/core/registry.c src/core/common.c src/core/docs.c
      src/query/query.c src/platform/linux.c)
    target_include_directories(inv_windows_schema_tests BEFORE PRIVATE ${PROJECT_BINARY_DIR}/schema-config)
    target_link_libraries(inv_windows_schema_tests PRIVATE inv_build_flags Threads::Threads m)
    if(INV_WITH_MEMORY AND INV_WITH_CPU AND
       INV_WITH_DMI AND INV_WITH_NETWORK AND INV_WITH_PROCESS AND INV_WITH_FILESYSTEM)
      file(GLOB windows_queries CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/queries/windows/*.invq")
      add_test(NAME windows_schema COMMAND inv_windows_schema_tests ${windows_queries})
    else()
      add_test(NAME windows_schema COMMAND inv_windows_schema_tests)
    endif()
  endif()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME documentation COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/embed_docs.py --check)
    if(WIN32)
      add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/windows_integration.py $<TARGET_FILE:inv>)
      if(INV_WITH_GUI)
        add_test(NAME windows_gui COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/windows_gui.py $<TARGET_FILE:inv>)
      endif()
    else()
      add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:inv>)
      if(INV_WITH_GUI)
        find_program(INV_XVFB_EXECUTABLE Xvfb)
        if(INV_XVFB_EXECUTABLE)
          add_test(NAME gui_x11 COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/gui_smoke.py $<TARGET_FILE:inv>)
        endif()
      endif()
      if(INV_WITH_TUI)
        add_test(NAME tui_pty COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/tui_pty.py $<TARGET_FILE:inv>)
      endif()
    endif()
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS inv RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY queries/ DESTINATION ${CMAKE_INSTALL_DATADIR}/inv/queries)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR})
message(STATUS "inv ${PROJECT_VERSION}: Windows=${INV_OS_WINDOWS}, GUI=${INV_WITH_GUI}, TUI=${INV_WITH_TUI}, static profile=${INV_STRICT_STATIC}")
