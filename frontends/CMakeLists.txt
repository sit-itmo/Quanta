add_library(quanta_output STATIC common/format.c common/view.c common/file.c)
quanta_target(quanta_output)
target_include_directories(quanta_output PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(quanta_output PRIVATE private ${PROJECT_SOURCE_DIR}/sensors/include)
target_link_libraries(quanta_output PUBLIC quanta_core)

if(QUANTA_WITH_TUI)
    if(WIN32)
        set(terminal tui/terminal_windows.c)
    else()
        set(terminal tui/terminal_posix.c)
    endif()
    add_library(quanta_tui STATIC tui/tui.c ${terminal})
    quanta_target(quanta_tui)
    target_include_directories(quanta_tui PRIVATE private ${PROJECT_SOURCE_DIR}/sensors/include)
    target_link_libraries(quanta_tui PUBLIC quanta_output)
endif()
if(QUANTA_WITH_GUI)
    enable_language(CXX)
    include(${PROJECT_SOURCE_DIR}/cmake/ImGui.cmake)
    if(WIN32)
        set(window gui/win32_platform.c)
    else()
        set(window gui/xcb_platform.c)
        find_path(QUANTA_XCB_INCLUDE_DIR xcb/xcb.h REQUIRED)
        find_library(QUANTA_XCB_LIBRARY xcb REQUIRED)
    endif()
    add_library(quanta_gui STATIC gui/gui.cpp gui/raster.c ${window})
    quanta_target(quanta_gui)
    target_compile_features(quanta_gui PRIVATE cxx_std_11)
    target_link_libraries(quanta_gui PUBLIC quanta_output PRIVATE imgui_vendor)
    if(WIN32)
        target_link_libraries(quanta_gui PRIVATE user32 gdi32 imm32)
    else()
        target_include_directories(quanta_gui PRIVATE ${QUANTA_XCB_INCLUDE_DIR})
        target_link_libraries(quanta_gui PRIVATE ${QUANTA_XCB_LIBRARY})
    endif()
endif()

# The entrypoints share the same parser/core; CSV and TXT differ only in default format.
foreach(app quanta quanta_cli quanta_csv quanta_txt)
    add_executable(${app} cli/main.c)
    quanta_target(${app})
    set_target_properties(${app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
    target_include_directories(${app} PRIVATE private)
    target_link_libraries(${app} PRIVATE quanta_output quanta_sensors)
    if(app STREQUAL "quanta_csv")
        target_compile_definitions(${app} PRIVATE QUANTA_DEFAULT_CSV=1)
    elseif(app STREQUAL "quanta_txt")
        target_compile_definitions(${app} PRIVATE QUANTA_DEFAULT_TXT=1)
    endif()
    if(QUANTA_WITH_TUI)
        target_link_libraries(${app} PRIVATE quanta_tui)
    endif()
    if(QUANTA_WITH_GUI)
        target_link_libraries(${app} PRIVATE quanta_gui)
    endif()
    if(WIN32)
        enable_language(RC)
        target_sources(${app} PRIVATE cli/windows_entry.c ${PROJECT_SOURCE_DIR}/resources/quanta.rc)
        target_include_directories(${app} PRIVATE ${PROJECT_SOURCE_DIR}/resources)
        if(MINGW)
            target_link_options(${app} PRIVATE -municode -Wl,--nxcompat -Wl,--dynamicbase)
        else()
            target_link_options(${app} PRIVATE /DYNAMICBASE /NXCOMPAT /MANIFEST:NO)
        endif()
    endif()
endforeach()
