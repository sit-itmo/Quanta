# frontends/CMakeLists.txt
# Shared frontend and exactly one GUI backend; backend libraries are outside the core.
add_library(quanta_frontends STATIC common/format.c common/view.c common/file.c dispatch.c)
quanta_target(quanta_frontends)
target_include_directories(quanta_frontends PRIVATE private)
target_link_libraries(quanta_frontends PUBLIC quanta_core)
if(WIN32)
    target_include_directories(quanta_frontends PRIVATE ${PROJECT_SOURCE_DIR}/osal/private)
endif()
if(QUANTA_WITH_TUI)
    if(WIN32)
        set(terminal tui/terminal_windows.c)
    else()
        set(terminal tui/terminal_posix.c)
    endif()
    target_sources(quanta_frontends PRIVATE tui/tui.c ${terminal})
endif()
if(QUANTA_WITH_GUI)
    include(${PROJECT_SOURCE_DIR}/cmake/ImGui.cmake)
    target_sources(quanta_frontends PRIVATE gui/gui.cpp)
    target_include_directories(quanta_frontends PRIVATE gui gui/backend)
    target_compile_features(quanta_frontends PRIVATE cxx_std_11)
    target_link_libraries(quanta_frontends PRIVATE imgui_vendor)
    if(QUANTA_GUI_BACKEND STREQUAL "OPENGL")
        if(WIN32)
            set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
            set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
            set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
            set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
            set(GLFW_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
            if(QUANTA_STRICT_STATIC)
                set(USE_MSVC_RUNTIME_LIBRARY_DLL OFF CACHE BOOL "" FORCE)
            else()
                set(USE_MSVC_RUNTIME_LIBRARY_DLL ON CACHE BOOL "" FORCE)
            endif()
            # Keep dependency examples/docs/tests outside the product build.
            # GLFW uses this compiler and the same MSVC runtime policy as Quanta.
            add_subdirectory("${QUANTA_GLFW_DIR}" "${PROJECT_BINARY_DIR}/vendor/glfw")
            message(STATUS "GLFW: building local static sources from ${QUANTA_GLFW_DIR}")
        endif()
        add_library(quanta_imgui_opengl STATIC
            ${QUANTA_IMGUI_DIR}/backends/imgui_impl_glfw.cpp
            ${QUANTA_IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
        target_link_libraries(quanta_imgui_opengl PUBLIC imgui_vendor glfw OpenGL::GL ${CMAKE_DL_LIBS})
        target_include_directories(quanta_imgui_opengl PUBLIC ${QUANTA_IMGUI_DIR}/backends)
        target_compile_features(quanta_imgui_opengl PRIVATE cxx_std_11)
        if(MSVC)
            target_compile_options(quanta_imgui_opengl PRIVATE /utf-8)
        endif()
        if(QUANTA_SANITIZE)
            target_compile_options(quanta_imgui_opengl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endif()
        target_sources(quanta_frontends PRIVATE gui/opengl.cpp)
        target_link_libraries(quanta_frontends PRIVATE quanta_imgui_opengl)
    else()
        target_sources(quanta_frontends PRIVATE gui/framebuffer.cpp)
        add_library(quanta_framebuffer STATIC gui/backend/imgui_renderer.cpp
                    gui/backend/raster.c gui/backend/portable_quad.cpp)
        quanta_target(quanta_framebuffer)
        target_include_directories(quanta_framebuffer PUBLIC gui/backend)
        target_link_libraries(quanta_framebuffer PUBLIC quanta_core imgui_vendor)
        target_link_libraries(quanta_frontends PRIVATE quanta_framebuffer)
        if(WIN32)
            target_sources(quanta_frontends PRIVATE gui/backend/windows.c)
            target_link_libraries(quanta_frontends PRIVATE user32 gdi32 imm32)
        else()
            target_sources(quanta_frontends PRIVATE gui/backend/x11.c)
            target_link_libraries(quanta_frontends PRIVATE X11::X11)
        endif()
    endif()
endif()
